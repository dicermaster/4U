<script>


    const texts = [
        "Привет, солнышко ❤️",
        "В общем пусть мы и не давно знакомы",
        "Ты стала чем-то особенным для меня",
        "Ты есть в каждом дне и я очень рад этому❤️",
        "Я очень рад что ты делишься всем со мной❤️",
        "Ты очень сильная красивая и умная девочка",
        "Я рад что среди всех встретил именно тебя",
	"Ты моя первая и последняя мысль утром и перед сном❤️",
	"Каждая твоя победа греет душу",
	"А каждое плохое настроение заставляет грустить и меня",
	"Ты большая умничка и я хочу что бы ты знала что я тобой горжусь❤️",
	"Как бы не было плохо или тяжело я всегда буду стараться поддержать и быть рядом❤️",
        "Я люблю тебя 💗"
    ];


    /*
    ============================================================
    НИЖЕ НИЧЕГО МЕНЯТЬ НЕ НУЖНО
    ============================================================
    */

    const textElement = document.getElementById("text");
    const button = document.getElementById("button");
    const page = document.getElementById("page");

    let currentText = 0;


    // Показываем новый текст
    function showNextText() {

        // Если тексты закончились — начинаем сначала
        currentText++;

        if (currentText >= texts.length) {
            currentText = 0;
        }

        // Плавно скрываем старый текст
        textElement.classList.add("hide");

        setTimeout(() => {

            textElement.textContent = texts[currentText];

            // Показываем новый
            textElement.classList.remove("hide");

        }, 350);
    }


    // Создание сердечка
    function createHeart(x, y) {

        const heart = document.createElement("div");

        heart.className = "heart";

        // Разные варианты сердечек
        const hearts = [
            "❤️",
            "💕",
            "💗",
            "💖",
            "💓",
            "💘"
        ];

        heart.textContent =
            hearts[Math.floor(Math.random() * hearts.length)];


        // Размер
        const scale =
            0.7 + Math.random() * 1.2;

        // Направление полёта
        const moveX =
            (Math.random() - 0.5) * 350;

        const moveY =
            -(100 + Math.random() * 300);

        const rotate =
            (Math.random() - 0.5) * 90;


        heart.style.left = x + "px";
        heart.style.top = y + "px";

        heart.style.setProperty(
            "--x",
            moveX + "px"
        );

        heart.style.setProperty(
            "--y",
            moveY + "px"
        );

        heart.style.setProperty(
            "--scale",
            scale
        );

        heart.style.setProperty(
            "--rotate",
            rotate + "deg"
        );


        // Немного разные размеры
        heart.style.fontSize =
            (20 + Math.random() * 25) + "px";


        page.appendChild(heart);


        // Удаляем после анимации
        setTimeout(() => {
            heart.remove();
        }, 2100);
    }


    // Создаём много сердечек
    function createHeartExplosion(x, y) {

        const amount = 25;

        for (let i = 0; i < amount; i++) {

            // Небольшая задержка делает эффект живее
            setTimeout(() => {

                createHeart(
                    x + (Math.random() - 0.5) * 40,
                    y + (Math.random() - 0.5) * 30
                );

            }, i * 15);
        }
    }


    // Вспышка
    function createFlash(x, y) {

        const flash = document.createElement("div");

        flash.className = "flash";

        flash.style.left = x + "px";
        flash.style.top = y + "px";

        page.appendChild(flash);

        setTimeout(() => {
            flash.remove();
        }, 700);
    }


    // Меняем фон
    function changeBackground() {

        const backgrounds = [
            "linear-gradient(135deg, #ffd6e7, #ffeaf3)",
            "linear-gradient(135deg, #ffb6d5, #ffd1e5)",
            "linear-gradient(135deg, #ff9fc8, #ffc4dd)",
            "linear-gradient(135deg, #ff8fbd, #ffb5d3)",
            "linear-gradient(135deg, #ff78ae, #ffa8cb)",
            "linear-gradient(135deg, #ff6ba5, #ff9bc3)"
        ];

        const index =
            Math.min(currentText, backgrounds.length - 1);

        document.body.style.background =
            backgrounds[index];
    }


    // Нажатие на кнопку
    button.addEventListener("click", function(event) {

        const rect = page.getBoundingClientRect();

        const x =
            event.clientX - rect.left;

        const y =
            event.clientY - rect.top;


        // Сердечки
        createHeartExplosion(x, y);

        // Вспышка
        createFlash(x, y);

        // Новый текст
        showNextText();

        // Новый фон
        changeBackground();

    });


    
    function createBackgroundHeart() {

        const heart =
            document.createElement("div");

        heart.className =
            "background-heart";

        const hearts = [
            "♡",
            "♥",
            "❤",
            "💕"
        ];

        heart.textContent =
            hearts[Math.floor(Math.random() * hearts.length)];

        heart.style.left =
            Math.random() * 100 + "%";

        heart.style.fontSize =
            (15 + Math.random() * 30) + "px";

        const duration =
            7 + Math.random() * 8;

        heart.style.animationDuration =
            duration + "s";

        page.appendChild(heart);

        setTimeout(() => {
            heart.remove();
        }, duration * 1000);
    }


 
    setInterval(
        createBackgroundHeart,
        700
    );

</script>
